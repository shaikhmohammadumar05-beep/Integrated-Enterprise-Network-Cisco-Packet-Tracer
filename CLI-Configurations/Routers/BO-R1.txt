 exit

! ===== ACLs =====
access-list 20 remark SSH-MANAGEMENT-ONLY

access-list 20 permit 100.0.0.0 0.0.0.255
access-list 20 permit 100.1.0.0 0.0.0.255
access-list 120 remark LAN-INBOUND-POLICY (whitelist)
access-list 120 permit udp any eq bootpc any eq bootps
access-list 120 permit ospf any any
access-list 120 deny tcp any any eq telnet
access-list 120 permit udp any any eq domain
access-list 120 permit udp any eq domain any
access-list 120 permit tcp any any eq domain
access-list 120 permit tcp any any eq www
access-list 120 permit tcp any any eq 443
access-list 120 permit tcp any any established

access-list 120 deny ip any any
